-- 1. CONFIGURAÇÕES INICIAIS E TIPOS
SET password_encryption = 'scram-sha-256';

CREATE TYPE tipo_usuario AS ENUM ('COMPRADOR', 'VENDEDOR');

-- 2. TABELAS DO PROJETO INGRESSEAI (Com constraints de segurança do PDF)
CREATE TABLE usuarios (
    id BIGSERIAL PRIMARY KEY,
    nome_completo VARCHAR(255) NOT NULL,
    cpf_cnpj VARCHAR(20) NOT NULL UNIQUE,
    cpf_cnpj_indice VARCHAR(64) UNIQUE,
    senha VARCHAR(255) NOT NULL,
    tipo tipo_usuario NOT NULL,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE eventos (
    id BIGSERIAL PRIMARY KEY,
    vendedor_id BIGINT NOT NULL,
    titulo VARCHAR(255) NOT NULL,
    descricao TEXT NOT NULL,
    tipo_evento VARCHAR(50) NOT NULL, 
    data_evento TIMESTAMP NOT NULL,
    quantidade_ingressos INTEGER CHECK (quantidade_ingressos >= 0), 
    quantidade_disponivel INTEGER CHECK (quantidade_disponivel >= 0),
    valor_ingresso NUMERIC(10,2) CHECK (valor_ingresso >= 0),
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (vendedor_id) REFERENCES usuarios(id)
);

CREATE TABLE ingressos (
    id BIGSERIAL PRIMARY KEY,
    evento_id BIGINT NOT NULL,
    quantidade INT NOT NULL CHECK (quantidade > 0 AND quantidade <= 4), 
    valor DECIMAL(10, 2) NOT NULL CHECK (valor >= 0), 
    FOREIGN KEY (evento_id) REFERENCES eventos(id)
);

CREATE TABLE compras (
    id BIGSERIAL PRIMARY KEY,
    comprador_id BIGINT NOT NULL,
    ingresso_id BIGINT NOT NULL,
    quantidade_comprada INT NOT NULL CHECK (quantidade_comprada > 0 AND quantidade_comprada <= 4),
    data_compra TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (comprador_id) REFERENCES usuarios(id),
    FOREIGN KEY (ingresso_id) REFERENCES ingressos(id)
);

-- 3. VIEWS DO PROJETO (Conceito de Ocultação do PDF)
CREATE OR REPLACE VIEW vw_usuarios_atendimento AS
SELECT id, nome_completo, tipo, criado_em
FROM usuarios;

CREATE ROLE atendimento LOGIN PASSWORD 'SenhaAtendimento123';

GRANT USAGE ON SCHEMA public TO app_runtime;
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLE public.usuarios, public.eventos, public.ingressos, public.compras TO app_runtime;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO app_runtime;

GRANT CONNECT ON DATABASE ingresse_ai TO atendimento;
GRANT USAGE ON SCHEMA public TO atendimento;
GRANT SELECT ON public.vw_usuarios_atendimento TO atendimento;

CREATE TABLE Transacoes (
    ID SERIAL PRIMARY KEY,
    Conta_Origem VARCHAR(50) NOT NULL,
    Conta_Destino VARCHAR(50) NOT NULL,
    Valor DECIMAL(15,2) NOT NULL,
    CONSTRAINT chk_valor_positivo CHECK (Valor > 0),
    CONSTRAINT chk_contas_diferentes CHECK (Conta_Origem <> Conta_Destino)
);
